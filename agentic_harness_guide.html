نسخه v1.py: معرفی Tool Schemas و فراخوانی ابزار

Tool Calling • Single Iteration

در این گام، مدل به ابزارهای سیستمی مجهز می‌شود. مدل مستقیماً کد پایتون اجرا نمی‌کند، بلکه اسکیمای توابع را می‌خواند و در قالب یک آرایه از اشیای tool_calls تقاضای اجرای آن را مطرح می‌کند. ما تابع پایتون را اجرا کرده و با نقش role: "tool" پاسخ را به مدل برمی‌گردانیم.

کد کامل فایل v1.py 
import json
from pathlib import Path
import sys
from openai import OpenAI

if sys.platform == "win32":
    sys.stdout.reconfigure(encoding="utf-8")

client = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")
MODEL = "qwen2.5:3b"
WORKSPACE_DIR = Path("workspace")
WORKSPACE_DIR.mkdir(exist_ok=True)

# ۱. تعریف ابزارهای پایتونی
def list_files():
    return [f.name for f in WORKSPACE_DIR.iterdir() if f.is_file()]

def read_file(file_name: str):
    file_path = WORKSPACE_DIR / file_name
    if not file_path.exists():
        return f"Error: File '{file_name}' not found."
    return file_path.read_text(encoding="utf-8")

tools = {"list_files": list_files, "read_file": read_file}

# ۲. تعریف کاتالوگ اسکیمای ابزارها برای مدل
tool_schemas = [
    {
        "type": "function",
        "function": {
            "name": "list_files",
            "description": "List all files in workspace directory.",
            "parameters": {"type": "object", "properties": {}},
        },
    },
    {
        "type": "function",
        "function": {
            "name": "read_file",
            "description": "Read content of a specific file.",
            "parameters": {
                "type": "object",
                "properties": {"file_name": {"type": "string"}},
                "required": ["file_name"],
            },
        },
    },
]

# ۳. مدیریت درخواست اجرای ابزار (تک مرحله‌ای)
def chat(prompt: str) -> str:
    messages = [
        {"role": "system", "content": "You are a helpful personal assistant."},
        {"role": "user", "content": prompt},
    ]

    response = client.chat.completions.create(
        model=MODEL,
        messages=messages,
        tools=tool_schemas,
    )
    message = response.choices[0].message

    if not message.tool_calls:
        return message.content or ""

    messages.append(message)

    for call in message.tool_calls:
        args = json.loads(call.function.arguments)
        name = call.function.name
        result = tools[name](**args) if name in tools else "Unknown tool"

        messages.append({
            "role": "tool",
            "tool_call_id": call.id,
            "content": str(result),
        })

    final = client.chat.completions.create(model=MODEL, messages=messages)
    return final.choices[0].message.content or ""

نسخه v2.py: ساخت حلقه عاملی خودکار (The Agentic Loop)

Autonomous • Multi-step

مشکل نسخه قبلی این بود که تنها یک دور ابزار صدا می‌زد. در دنیای واقعی، ایجنت باید بتواند ابتدا فایل‌ها را لیست کند، سپس یکی را بخواند و بعد خروجی را در فایلی دیگر ذخیره کند. این قابلیت با حلقه پیوسته while True و بررسی شرط خروج ایجاد می‌شود.

کد بخش حلقه خودکار در v2.py 
def chat(prompt: str) -> str:
    messages = [
        {"role": "system", "content": "You are a helpful personal assistant."},
        {"role": "user", "content": prompt},
    ]

    # چرخه عاملی پیوسته (Agentic Loop)
    while True:
        response = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=tool_schemas,
        )
        message = response.choices[0].message

        # شرط توقف: مدل تصمیم گرفته نیازی به ابزار ندارد و پاسخ نهایی را تولید کرده است
        if not message.tool_calls:
            return message.content or ""

        # ثبت پیام مدل برای حفظ کانتکست
        messages.append(message)

        # اجرای تمامی ابزارهای درخواست شده در این مرحله
        for call in message.tool_calls:
            args = json.loads(call.function.arguments)
            name = call.function.name
            print(f"[Loop Action] Calling {name} with {args}")
            
            result = tools[name](**args) if name in tools else "Tool not found"
            
            # بازگرداندن نتیجه به کانتکست
            messages.append({
                "role": "tool",
                "tool_call_id": call.id,
                "content": str(result),
            })
        # حلقه ادامه می‌یابد تا مدل نتیجه ابزار را مشاهده و اقدام بعدی را تعیین کند

import asyncio
from contextlib import AsyncExitStack
import json
import os
from pathlib import Path
import sys

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client
from openai import AsyncOpenAI

if sys.platform == "win32":
    sys.stdout.reconfigure(encoding="utf-8")

ROOT_DIR = Path(__file__).parent if "__file__" in locals() else Path.cwd()
WORKSPACE_DIR = ROOT_DIR / "workspace"
MEMORY_FILE = ROOT_DIR / "memory.md"
MCP_CONFIG_FILE = ROOT_DIR / "mcp_servers.json"

WORKSPACE_DIR.mkdir(exist_ok=True)

MODEL_CONFIGS = {
    "local": {"model": "qwen2.5:3b", "base_url": "http://localhost:11434/v1", "api_key": "ollama"},
    "local_small": {"model": "qwen2.5:1.5b", "base_url": "http://localhost:11434/v1", "api_key": "ollama"},
    "openrouter": {"model": "anthropic/claude-3.5-sonnet", "base_url": "https://openrouter.ai/api/v1", "api_key": os.getenv("OPENROUTER_API_KEY", "")},
}

current_model_key = "local"

def get_client(model_key: str):
    conf = MODEL_CONFIGS[model_key]
    return AsyncOpenAI(base_url=conf.get("base_url"), api_key=conf.get("api_key") or "dummy"), conf["model"]

def load_memory() -> str:
    return MEMORY_FILE.read_text(encoding="utf-8") if MEMORY_FILE.exists() else ""

def save_memory(content: str) -> str:
    MEMORY_FILE.write_text(content, encoding="utf-8")
    return "Memory successfully updated."

def list_files():
    return [f.name for f in WORKSPACE_DIR.iterdir() if f.is_file()]

def read_file(file_name: str):
    p = WORKSPACE_DIR / file_name
    return p.read_text(encoding="utf-8") if p.exists() else f"File {file_name} not found"

def write_file(file_name: str, content: str):
    (WORKSPACE_DIR / file_name).write_text(content, encoding="utf-8")
    return f"File '{file_name}' saved."

local_tools = {"list_files": list_files, "read_file": read_file, "write_file": write_file, "save_memory": save_memory}

local_tool_schemas = [
    {"type": "function", "function": {"name": "list_files", "description": "List files", "parameters": {"type": "object", "properties": {}}}},
    {"type": "function", "function": {"name": "read_file", "description": "Read file", "parameters": {"type": "object", "properties": {"file_name": {"type": "string"}}, "required": ["file_name"]}}},
    {"type": "function", "function": {"name": "write_file", "description": "Write file", "parameters": {"type": "object", "properties": {"file_name": {"type": "string"}, "content": {"type": "string"}}, "required": ["file_name", "content"]}}},
    {"type": "function", "function": {"name": "save_memory", "description": "Save to memory", "parameters": {"type": "object", "properties": {"content": {"type": "string"}}, "required": ["content"]}}},
]

mcp_sessions = {}
all_tool_schemas = list(local_tool_schemas)
exit_stack = AsyncExitStack()

async def connect_mcp_servers():
    if not MCP_CONFIG_FILE.exists():
        return
    with open(MCP_CONFIG_FILE, "r", encoding="utf-8") as f:
        servers = json.load(f).get("mcpServers", {})

    for name, s_info in servers.items():
        params = StdioServerParameters(command=s_info["command"], args=s_info.get("args", []))
        r, w = await exit_stack.enter_async_context(stdio_client(params))
        session = await exit_stack.enter_async_context(ClientSession(r, w))
        await session.initialize()
        res = await session.list_tools()
        for t in res.tools:
            mcp_sessions[t.name] = session
            all_tool_schemas.append({"type": "function", "function": {"name": t.name, "description": t.description or "", "parameters": t.inputSchema}})

async def call_tool(name: str, args: dict):
    if name in local_tools:
        return local_tools[name](**args)
    if name in mcp_sessions:
        r = await mcp_sessions[name].call_tool(name, arguments=args)
        texts = [i.text for i in r.content if getattr(i, "type", "") == "text"]
        return "\n".join(texts) if texts else str(r.content)
    return f"Error: Tool {name} not found"

async def run_agent(prompt: str) -> str:
    client, model = get_client(current_model_key)
    system_prompt = f"You are a helpful assistant.\n\nMemory:\n{load_memory()}"
    messages = [{"role": "system", "content": system_prompt}, {"role": "user", "content": prompt}]

    while True:
        resp = await client.chat.completions.create(model=model, messages=messages, tools=all_tool_schemas)
        msg = resp.choices[0].message
        if not msg.tool_calls:
            return msg.content or ""
        messages.append(msg)
        for call in msg.tool_calls:
            args = json.loads(call.function.arguments)
            result = await call_tool(call.function.name, args)
            messages.append({"role": "tool", "tool_call_id": call.id, "content": str(result)})

def handle_slash_command(cmd: str) -> bool:
    global current_model_key
    parts = cmd.strip().split()
    b = parts[0].lower()
    if b in ("/q", "/quit"): return False
    if b == "/models":
        print("\n".join([f" - {k}: {v['model']}{' (Active)' if k == current_model_key else ''}" for k, v in MODEL_CONFIGS.items()]))
    elif b == "/model" and len(parts) > 1 and parts[1] in MODEL_CONFIGS:
        current_model_key = parts[1]
        print(f"Model switched to {current_model_key}")
    elif b == "/tools":
        print("\n".join([f" - {s['function']['name']}" for s in all_tool_schemas]))
    elif b == "/memory":
        print(load_memory() or "(Empty)")
    return True

async def main():
    try:
        await connect_mcp_servers()
        print("Harness ready. Commands: /models, /model , /tools, /memory, /quit")
        while True:
            u = await asyncio.to_thread(input, f"[{current_model_key}] You: ")
            if not u.strip(): continue
            if u.startswith("/"):
                if not handle_slash_command(u): break
                continue
            print(f"\nAssistant:\n{await run_agent(u)}\n")
    finally:
        await exit_stack.aclose()

if __name__ == "__main__":
    asyncio.run(main())

v2.py همگام (Sync) while True پیوسته خواندن و نوشتن فایل محدود به سشن استدلال چندمرحله‌ای خودکار
v3.py ناهمگام (Asyncio) while True ناهمگام پروتکل MCP و وب محدود به سشن کشف داینامیک ابزارها (Discovery)
v4.py همگام/ناهمگام while True پیوسته فایل + ابزار حافظه پایدار در memory.md ماندگاری داده‌ها بین نشست‌ها
Harness مسیریابی ماژولار حلقه کامل ایجنتی ترکیبی (MCP + Local) پایدار و هوشمند کماندهای ترمینال و سوییچ مدل

تعیین سقف تکرار (Max Iterations)

در حلقه while True همیشه متغیر step_count < 15 را کنترل کنید تا اگر مدل دچار توهم یا لوپ باطل شد، توکن‌ها و بودجه پروژه هدر نرود.

🛡️

تایید انسانی (Human-in-the-Loop)

برای ابزارهای پرریسک نظیر حذف فایل، تغییر دیتابیس یا اجرای شل، حتماً قبل از اجرای اکشن در پایتون یک ورودی تأییدیه صریح از کاربر دریافت نمایید.

🧹

هرس و تلخیص کانتکست (Context Pruning)

اگر تسک شامل ۲۰ دور فراخوانی ابزار است، تاریخچه پیام‌ها سرریز خواهد شد؛ نتایج طولانی ابزارها را پس از خوانده شدن توسط مدل خلاصه و فشرده کنید.
